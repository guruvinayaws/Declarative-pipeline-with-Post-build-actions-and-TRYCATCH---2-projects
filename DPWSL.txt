@Library('mylibrary')_
pipeline
{
    agent any
    stages
    {
        stage('Download')
        {
            steps
            {
                  script
                 {
                   cicd.gitDownload("maven")
                 }
            }
        }
       stage('Build')
       {
            steps
            {
                script
               {
                  cicd.buildArtifact()
               }
            }
       }
       stage('Deploy')
        {
           steps
             {
                script
                {
                   cicd.deployTomcat("172.31.71.255","testapp")
                }
             }
        }
        stage('Test')
        {
            steps
            {
                script
                {
                    cicd.gitDownload("FunctionalTesting")
                    cicd.runSelenium()
                }
            }
        }
        stage('Delivery')
        {
            steps
            {
                script
                {
                    cicd.deployTomcat("172.31.64.9","prodapp")
                }
            }
        }
    }
}
